include(FetchContent)

FetchContent_Declare(
  allwpilib
  GIT_REPOSITORY https://github.com/wpilibsuite/allwpilib
  GIT_TAG v2025.3.2
)

# These options should be set before FetchContent_MakeAvailable
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_JAVA OFF CACHE BOOL "" FORCE)
set(WITH_DOCS OFF CACHE BOOL "" FORCE)
set(WITH_CSCORE OFF CACHE BOOL "" FORCE)
set(WITH_NTCORE OFF CACHE BOOL "" FORCE)
set(WITH_WPICAL OFF CACHE BOOL "" FORCE)
set(WITH_WPIMATH OFF CACHE BOOL "" FORCE)
set(WITH_WPIUNITS OFF CACHE BOOL "" FORCE)
set(WITH_WPILIB OFF CACHE BOOL "" FORCE)
set(WITH_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_GUI OFF CACHE BOOL "" FORCE)
set(WITH_PROTOBUF OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_FMTLIB OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_LIBUV OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_EIGEN OFF CACHE BOOL "" FORCE)
set(NO_WERROR OFF CACHE BOOL "" FORCE)
set(WITH_SIMULATION_MODULES OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(allwpilib)
